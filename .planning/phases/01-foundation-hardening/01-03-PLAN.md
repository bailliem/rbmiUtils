   ```r
   # Compute digest of the original imputed_data for round-trip verification
   # Store column metadata needed for verification
   col_metadata <- lapply(names(imputed_data), function(cn) {
     list(
       class = class(imputed_data[[cn]]),
       typeof = typeof(imputed_data[[cn]])
     )
   })
   names(col_metadata) <- names(imputed_data)

   # Create a representative sample for digest (first imputation only, for efficiency)
   first_imp <- imputed_data[imputed_data$IMPID == unique(imputed_data$IMPID)[1], , drop = FALSE]
   digest_val <- rlang::hash(list(
     col_metadata = col_metadata,
     first_imp_hash = rlang::hash(first_imp),
     n_cols = ncol(imputed_data),
     col_names = names(imputed_data),
     n_imps = length(unique(imputed_data$IMPID))
   ))

   attr(reduced, "rbmiUtils_digest") <- digest_val
   attr(reduced, "rbmiUtils_col_metadata") <- col_metadata
   attr(reduced, "rbmiUtils_col_names") <- names(imputed_data)
   ```

2. In `expand_imputed_data()`, after constructing the result (line 308), add digest verification:

   ```r
   # Verify round-trip integrity if digest is available
   stored_digest <- attr(reduced_data, "rbmiUtils_digest")
   if (!is.null(stored_digest)) {
     stored_col_meta <- attr(reduced_data, "rbmiUtils_col_metadata")
     stored_col_names <- attr(reduced_data, "rbmiUtils_col_names")

     # Check column names match
     if (!identical(sort(intersect(names(result), stored_col_names)), sort(stored_col_names))) {
       missing_in_result <- setdiff(stored_col_names, names(result))
       extra_in_result <- setdiff(names(result), stored_col_names)
       cli::cli_abort(
         c(
           "Round-trip integrity check failed: column mismatch.",
           if (length(missing_in_result) > 0) paste0("Missing columns: ", paste(missing_in_result, collapse = ", ")),
           if (length(extra_in_result) > 0) paste0("Extra columns: ", paste(extra_in_result, collapse = ", "))
         ),
         class = "rbmiUtils_error_integrity"
       )
     }

     # Check column types match
     for (cn in stored_col_names) {
       if (cn %in% names(result) && !is.null(stored_col_meta[[cn]])) {
         expected_class <- stored_col_meta[[cn]]$class
         actual_class <- class(result[[cn]])
         if (!identical(expected_class, actual_class)) {
           cli::cli_abort(
             c(
               "Round-trip integrity check failed: column type mismatch.",
               "Column {.field {cn}}: expected {.cls {expected_class}}, got {.cls {actual_class}}."
             ),
             class = "rbmiUtils_error_integrity"
           )
         }
       }
     }
   }
   ```
